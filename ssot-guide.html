🔌 วิธีเชื่อมโปรแกรมเข้ากับราคากลาง

ทำครั้งเดียว · หลังจากนี้ทุกโปรแกรมจะได้ราคาใหม่อัตโนมัติทุกครั้งที่คุณแก้ที่เดียว

สรุปสั้นที่สุด — ใส่ script 1 บรรทัด แล้วเรียก SPPrices.load() เท่านั้น

ขั้นที่ 1 · ใส่ script

วางบรรทัดนี้ก่อน </body> ของโปรแกรมที่ต้องการ

<script src="https://nny-spcompetitors.netlify.app/sp-prices.js"></script>

ขั้นที่ 2 · ดึงราคาไปใช้

<script>
SPPrices.load().then(function (db) {

  // ราคาขายของแต่ละ SKU
  var a = db.price('สายฝน 80×200', 'de');   // 1349
  var b = db.price('วงกบประตู 90×200', 'tb'); // 1089

  // ค่าส่ง
  var ship = db.shippingOf('de', 'door');      // 250

  // ทั้งหมดในหมวด
  db.byCat('วงกบประตู').forEach(function (r) {
    console.log(r.sku, r.de, r.tb);
  });

  // รายชื่อหมวดทั้งหมด
  console.log(db.cats());

  // เอาไปวาดหน้าจอต่อได้เลย
  drawMyProgram(db);
});
</script>

ขั้นที่ 3 · กันโปรแกรมพัง (แนะนำ)

ถ้าเน็ตล่ม ให้ใช้สำเนาสำรองที่ฝังไว้ในโปรแกรม โปรแกรมจะไม่มีวางเปล่า

var FALLBACK = { /* วาง prices.json ที่ดาวน์โหลดมาตรงนี้ */ };

SPPrices.load({ fallback: FALLBACK })
  .then(function (db) {
    SPPrices.statusBadge(db);  // แสดงป้ายบอกว่าราคาสดหรือเก่า
    drawMyProgram(db);
  });

ฟังก์ชันที่ใช้ได้

คำสั่งคืนค่า
db.price(sku, shop)ราคาของ SKU · shop = 'de' หรือ 'tb' · ไม่มีคืน null
db.byCat(cat)รายการทั้งหมดในหมวดนั้น
db.cats()รายชื่อหมวดทั้งหมด
db.shippingOf(shop, kind)ค่าส่ง · kind = door / frame / round_frame
db.oursข้อมูลดิบทั้งหมด (array)
db.updatedวันที่อัปเดตล่าสุด
db.staletrue = ใช้สำเนาเก่า · false = สดจากเน็ต

วิธีอัปเดตราคา

ทางขั้นตอน
แก้เองเปิด หน้าแก้ราคากลาง → แก้ → ดาวน์โหลด prices.json → วางแทนที่ใน GitHub
ให้ Claude ทำเปิดหน้าแก้ราคา → แก้ → กด คัดลอกให้ Claude → วางในแชท
บอกในแชทตรงๆพิมพ์ราคาใหม่มาได้เลย ผมอัปเดต prices.json ให้ทันที
ข้อควรรู้
• ไฟล์ราคาตั้ง no-cache ไว้แล้ว โปรแกรมจะเห็นราคาใหม่ภายในไม่กี่วินาทีหลัง deploy
• ตัวเชื่อมเก็บ cache ในเครื่อง 5 นาที เพื่อไม่ให้ยิงถี่เกินไป · ถ้าอยากดึงสดทันทีใช้ SPPrices.load({force:true})
• ชื่อ SKU ต้องตรงกับใน prices.json เป๊ะ (ดูรายชื่อได้จากหน้าแก้ราคา)

ทดสอบว่าเชื่อมสำเร็จ

เปิด Console ในเบราว์เซอร์แล้วพิมพ์:

SPPrices.load().then(db => console.table(db.ours.slice(0,10)))
